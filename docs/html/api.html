API Reference

RB_Nexus 0.2.1 · ESP32-WROOM-32 · Arduino-ESP32 3.3.10

ใช้ #include <RB_Nexus.h> และ object RB; RB_Nexus เป็น alias ของ object เดียวกัน

ระบบหลัก

bool begin(uint32_t i2cFreq = 400000, uint32_t spiFreq = 1000000);
void update();
void setLED(bool on);
void toggleLED();

เริ่ม Serial.begin(115200) แยกต่างหาก; begin() คืนสถานะพบ PCA9685 แม้ไม่พบ PCA ก็ยังลอง IMU ได้

มอเตอร์และ PID

RB.motorSet(4, 128); // -255..255; ค่านอกช่วงถูก clamp
RB.motorStop(4);     // หยุดและปิด PID ช่องนี้
RB.motorStopAll();   // หยุดทุกช่องและปิด PID
RB.motorSetPID(4, 1.2f, 0.5f, 0.05f);
RB.motorSetRPM(4, 120); // เปิด PID; ต้อง encoderBegin() และ update()
RB.motorPIDDisable(4);
RB.emergencyStop(); // latch ป้องกัน motorSet/motorBrake
RB.clearEmergencyStop();

motorSet() เป็น open-loop จึงปิด PID ช่องนั้น; motorSetRPM(..., 0) หยุดและปิด PID; emergencyStop มีผลต่อมอเตอร์ผ่าน API มอเตอร์ ไม่ใช่วงจรหยุดฉุกเฉินอิสระ และ raw setPWM ยังเป็น API ระดับต่ำ

Encoder

RB.encoderBegin();
RB.encoderSetPPR(4, 330); // จำนวน ticks ที่นับจริงต่อรอบล้อ
int32_t ticks = RB.encoderRead(4);
float rpm = RB.encoderRPM(4);
RB.encoderReset(4);

นับ CHANGE ของ A (2 edges ต่อคาบ A) และอ่าน B เพื่อหาทิศทาง ต้องตั้ง PPR เป็นจำนวน counts ต่อรอบที่นับจริง RPM มี low-pass filter และอัปเดตทุกประมาณ 50 ms

Analog, Servo และ PWM

uint16_t raw = RB.analogRead(1); // A1..A8 ใช้ 1..8 เท่านั้น
float volts = RB.analogVoltage(1, 3.3f);
RB.servoWrite(8, 90);           // CH8..15, มุม 0..180
RB.servoWriteMicroseconds(8, 1500); // clamp 500..2500; 0 = detach
RB.servoDetach(8);
RB.pwmWrite(RB_PIN_D4, 128, 5000, 8);
RB.pwmStop(RB_PIN_D4);

PCA9685 ทุกช่องแชร์ความถี่เดียวกัน ค่าเริ่มต้นประมาณ 50 Hz เหมาะกับ RC servo การเปลี่ยน setPWMFreq() มีผลต่อทุกช่อง; pulse servo ที่สั่งใหม่คำนวณตามความถี่ที่ตั้งจริง

Watchdog

RB.enableWatchdog(2000);
RB.feedWatchdog();
RB.update();
bool stopped = RB.isEmergencyStopped();

เป็น software watchdog ต้องเรียก update() จึงตรวจ timeout ได้ ไม่ทำงานระหว่างโปรแกรมค้าง ห้าม feed ทุก loop หากต้องการตรวจการขาดการสื่อสาร ให้ feed เมื่อได้รับคำสั่งที่ถูกต้อง
